# Ubuntu Terminal

```bash
# ------------------------------------------------
# BASIC COMMANDS
# ------------------------------------------------

cd
cd ..
cd /path/to/directory

ls
ls -l
ls -a
ls -lh

mkdir new_directory
mkdir -p parent_directory/child_directory

rm file.txt
rm -r directory_to_remove

cp source_file.txt destination_file.txt
cp -r source_directory destination_directory

mv old_file.txt new_file.txt

rename old_file.txt new_file.txt



# ------------------------------------------------
# TREE
# ------------------------------------------------

tree

# List files and directories up to a depth of 2
tree -L 2

# List only directories up to a depth of 2
tree -L 2 -d

# Ignore node_modules, venv, and hidden files/directories
tree -I 'node_modules|venv|.*'
```
